816:    /// 3D float vector
817:    /// </summary>
818:    public struct Vector3
819:    {
820:        public float X { get; set; }
821:        public float Y { get; set; }
822:        public float Z { get; set; }
823:        
824:        public Vector3(float x, float y, float z)
825:        {
826:            X = x;
827:            Y = y;
828:            Z = z;
829:        }
830:        
831:        public static Vector3 operator +(Vector3 a, Vector3 b)
832:        {
833:            return new Vector3(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
834:        }
835:        
836:        public static Vector3 operator -(Vector3 a, Vector3 b)
837:        {
838:            return new Vector3(a.X - b.X, a.Y - b.Y, a.Z - b.Z);
839:        }
840:        
841:        public static Vector3 operator *(Vector3 a, float scalar)
842:        {
843:            return new Vector3(a.X * scalar, a.Y * scalar, a.Z * scalar);
844:        }
845:        
846:        public static float Distance(Vector3 a, Vector3 b)
847:        {
848:            var dx = a.X - b.X;
849:            var dy = a.Y - b.Y;
850:            var dz = a.Z - b.Z;
851:            return (float)Math.Sqrt(dx * dx + dy * dy + dz * dz);
852:        }
853:        
854:        public static Vector3 Normalize(Vector3 vector)
855:        {
856:            var length = (float)Math.Sqrt(vector.X * vector.X + vector.Y * vector.Y + vector.Z * vector.Z);
857:            if (length < 0.0001f)
858:                return Vector3.Zero;
859:                
860:            return new Vector3(vector.X / length, vector.Y / length, vector.Z / length);
861:        }
862:        
863:        public static Vector3 Zero => new Vector3(0, 0, 0);
864:        
865:        public override string ToString()
866:        {
867:            return $"({X:F2}, {Y:F2}, {Z:F2})";
868:        }
869:    }
